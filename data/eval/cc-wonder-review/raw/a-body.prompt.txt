269: export const MANTLE_RUN = 0.4;
270: const MANTLE_TICKS = 72;
271: export function wallPath(result: WallResult, x0: number, height: number): { x: number; y: number }[] {
272:   if (!result.grabbed) return result.path;
273:   const up = result.path.slice(0, result.grabTick! + 1);
274:   const from = up[up.length - 1];
275:   const to = { x: height + MANTLE_RUN, y: height };
276:   const out = [...up];
277:   for (let i = 1; i <= MANTLE_TICKS; i++) {
278:     const t = i / MANTLE_TICKS;
279:     const e = t * t * (3 - 2 * t); // smoothstep: slow start and end
280:     out.push({ x: from.x + (to.x - from.x) * e, y: from.y + (to.y - from.y) * e });
281:   }
282:   void x0;
283:   return out;
284: }

54: // One fixed step of 1/120 s. Every update below is the exact solution of its law over one step,
55: // so the closed forms in the tests hold at any step size, not just in the limit.
56: export function step(s: RunnerState, input: RunnerInput, p: RunnerParams, terrain: Terrain = FLAT): RunnerState {
57:   if (s.splashed) return { ...s, tick: s.tick + 1, vx: 0, vy: 0 };

59:   if (s.grounded) {

61:     // Without a push the feet skid: braking is the same grip-limited law.

63:     return onGround(dir !== 0 ? sprint(next, p, dir) : skid(next, p), s, p, terrain);
64:   }
65:   return fly(next, input, p, terrain);

73: // dv/dt = (top - v) / tau, so v(t) = top - (top - v0) e^(-t/tau).
74: function sprint(s: RunnerState, p: RunnerParams, dir: 1 | -1 = 1): RunnerState {
75:   const tau = p.topSpeed / p.acceleration;
76:   const decay = Math.exp(-DT / tau);
77:   const target = dir * p.topSpeed;
78:   const vx = target - (target - s.vx) * decay;
79:   const x = s.x + target * DT - (target - s.vx) * tau * (1 - decay);
80:   return { ...s, vx, x };
81: }

83: // Coulomb friction: constant deceleration grip * g until the feet stop.
84: function skid(s: RunnerState, p: RunnerParams): RunnerState {
85:   const decel = p.grip * p.gravity;
86:   const sign = s.vx < 0 ? -1 : 1;
87:   const speed = Math.abs(s.vx);
88:   const stopTime = speed / decel;
89:   if (stopTime >= DT) return { ...s, vx: sign * (speed - decel * DT), x: s.x + sign * (speed * DT - 0.5 * decel * DT * DT) };
90:   return { ...s, vx: 0, x: s.x + sign * 0.5 * speed * stopTime };
91: }

96: function onGround(moved: RunnerState, before: RunnerState, p: RunnerParams, t: Terrain): RunnerState {
97:   const here = groundAt(t, moved.x);
98:   const nowOn = platformAt(t, moved.x), wasOn = platformAt(t, before.x);
99:   if (nowOn?.profile || wasOn?.profile) {
100:     // on a curved top: too steep to run up means stopping where it becomes that steep (found by halving the step)
101:     const climbs = (x: number) => groundAt(t, x) - before.y > MAX_SLOPE * Math.abs(x - before.x) + 1e-9;
102:     if (climbs(moved.x)) {
103:       let lo = before.x, hi = moved.x;
104:       for (let i = 0; i < 30; i++) { const mid = (lo + hi) / 2; if (climbs(mid)) hi = mid; else lo = mid; }
105:       return { ...moved, x: lo, y: before.y, vx: 0 };
106:     }
107:   }
108:   if (here > before.y + p.stepUp) {
109:     const block = nowOn;
110:     if (!block) return { ...moved, x: before.x, vx: 0 };
111:     return { ...moved, x: moved.x < before.x ? block.x1 + 1e-6 : block.x0 - 1e-6, vx: 0 };
112:   }
113:   if (here < before.y - 1e-9) {
114:     // down a curved top the feet stay on the ground as long as the slope is gentler than 45 degrees: running over a dome is not
115:     // leaving the ground (so Space still works on the way down); a real drop (an edge) still lets go
116:     const onCurve = nowOn?.profile !== undefined && wasOn?.profile !== undefined;
117:     if (onCurve && before.y - here <= MAX_SLOPE * Math.abs(moved.x - before.x) + 1e-9) return { ...moved, y: here };
118:     return { ...moved, grounded: false, vy: 0 }; // off the edge: fall
119:   }
120:   return { ...moved, y: here };
121: }

123: // Gravity only. Constant acceleration, so x += v dt and y += v dt - g dt²/2 are exact.
124: function fly(s: RunnerState, input: RunnerInput, p: RunnerParams, t: Terrain): RunnerState {
125:   const vy = s.vy - p.gravity * DT;
126:   const y = s.y + s.vy * DT - 0.5 * p.gravity * DT * DT;
127:   let x = s.x + s.vx * DT;
128:   let vx = s.vx;
129:   const ground = groundAt(t, x);
130:   // Flew into the side of a block that stands higher than the feet: stop against it and drop.
131:   if (ground > s.y + 1e-9 && groundAt(t, s.x) < ground) {
132:     const block = platformAt(t, x);
133:     if (block && !block.profile) { x = s.vx < 0 ? block.x1 + 1e-6 : block.x0 - 1e-6; vx = 0; } // a curved top has no side to hit: rising ground is met by landing on it
134:   }
135:   const floor = groundAt(t, x);
136:   if (y > floor) {
137:     if (y < t.waterY) return { ...s, x, vx, y, vy, splashed: true };
138:     return { ...s, x, vx, y, vy };
139:   }
140:   // Touchdown inside this step: find the exact moment so range doesn't depend on the step size.
141:   const curved = platformAt(t, x)?.profile !== undefined || platformAt(t, s.x)?.profile !== undefined;
142:   if (curved) {
143:     // over a curved top the floor changes during the step: find the moment the path meets it by halving the time
144:     let lo = 0, hi = DT;
145:     for (let i = 0; i < 30; i++) {
146:       const tm = (lo + hi) / 2;
147:       const ym = s.y + s.vy * tm - 0.5 * p.gravity * tm * tm;
148:       if (ym > groundAt(t, s.x + s.vx * tm)) lo = tm; else hi = tm;
149:     }
150:     const tHit = hi, landedAtX = s.x + s.vx * tHit, landY = groundAt(t, landedAtX);
151:     const vyHit = s.vy - p.gravity * tHit;
152:     const report = landing(Math.max(0, -vyHit), input.squat, p.landing, !!input.roll);
153:     return {
154:       ...s, x: landedAtX, y: landY,
155:       vx: afterLanding(report, vx, p, !!platformAt(t, landedAtX)?.bouncy),
156:       vy: 0, grounded: true, landedAtTick: s.tick - 1 + tHit / DT, landedAtX, landing: report,
157:     };
158:   }
159:   const disc = Math.sqrt(s.vy * s.vy + 2 * p.gravity * (s.y - floor));
160:   const tHit = (s.vy + disc) / p.gravity;
161:   const landedAtX = s.x + s.vx * tHit;
162:   const report = landing(-(s.vy - p.gravity * tHit), input.squat, p.landing, !!input.roll);
163:   return {
164:     ...s,
165:     x: landedAtX,
166:     y: floor,
167:     vx: afterLanding(report, vx, p, !!platformAt(t, landedAtX)?.bouncy), // a trampoline catches the fall
168:     vy: 0,
169:     grounded: true,
170:     landedAtTick: s.tick - 1 + tHit / DT, // s.tick already counts this step
171:     landedAtX,
172:     landing: report,
173:   };
174: }

19: // Height of one platform's standing surface at x (flat unless it has a profile).
20: export function heightAt(p: Platform, x: number): number {
21:   const pr = p.profile;
22:   if (!pr || pr.length < 2) return p.top;
23:   if (x <= pr[0][0]) return pr[0][1];
24:   for (let i = 1; i < pr.length; i++) {
25:     if (x <= pr[i][0]) { const [xa, ya] = pr[i - 1], [xb, yb] = pr[i]; return xb === xa ? yb : ya + ((yb - ya) * (x - xa)) / (xb - xa); }
26:   }
27:   return pr[pr.length - 1][1];
28: }

30: // Highest standing surface at x, or -Infinity over a gap.
31: export function groundAt(t: Terrain, x: number): number {
32:   let top = -Infinity;
33:   for (const p of t.platforms) if (x >= p.x0 && x <= p.x1) top = Math.max(top, heightAt(p, x));
34:   return top;
35: }

37: // The platform whose left edge is the next one at or after x (used when the runner hits a block's side).
38: export function platformAt(t: Terrain, x: number): Platform | undefined {

40:   for (const p of t.platforms) {
41:     if (x < p.x0 || x > p.x1) continue;
42:     const h = p.profile ? heightAt(p, x) : p.top;
43:     if (h > bestH) { best = p; bestH = h; }
44:   }
45:   return best;
46: }

1: // Landing: the same speed change spread over more time is a gentler force (impulse = force x time).
2: export interface LandingParams {

23:   squatted: boolean;
24:   stopTime: number;
25:   peakDecel: number; // m/s², impactSpeed / stopTime (force per unit mass)
26:   stumbled: boolean;

29: }

31: export type LandStyle = 'stiff' | 'squat' | 'roll';
32: 
33: // The time a stop takes: the usual time for a soft landing, but never longer than the body can travel at constant push (distance = speed x time / 2).
34: function stopTimeFor(style: LandStyle, v: number, p: LandingParams): number {
35:   const time = style === 'roll' ? p.rollStopTime : style === 'squat' ? p.squatStopTime : p.stiffStopTime;
36:   const travel = style === 'roll' ? p.rollTravel : style === 'squat' ? p.squatTravel : p.stiffTravel;
37:   return travel === undefined || v <= 0 ? time : Math.min(time, (2 * travel) / v);
38: }

40: export function landing(impactSpeed: number, squatted: boolean, p: LandingParams = DEFAULT_LANDING, rolled = false): LandingReport {
41:   const style: LandStyle = rolled ? 'roll' : squatted ? 'squat' : 'stiff';
42:   const stopTime = stopTimeFor(style, impactSpeed, p);
43:   const peakDecel = impactSpeed / stopTime;
44:   const hurt = peakDecel > p.hurtDecel;
45:   return { impactSpeed, squatted: squatted && !rolled, stopTime, peakDecel, stumbled: !rolled && !hurt && peakDecel > p.stumbleDecel, rolled, hurt };
46: }

48: // The biggest fall each landing style survives, in metres (impact speed v = sqrt(2 g h)): worked out from the same stop times, not a table.
49: // `clean` is the biggest fall with no stumble (a roll never stumbles); `survive` is the biggest fall that does not end the run.
50: export function safeDrop(style: LandStyle, p: LandingParams = DEFAULT_LANDING, g = 9.81): { clean: number; survive: number } {
51:   const time = style === 'roll' ? p.rollStopTime : style === 'squat' ? p.squatStopTime : p.stiffStopTime;
52:   const travel = style === 'roll' ? p.rollTravel : style === 'squat' ? p.squatTravel : p.stiffTravel;
53:   const speedFor = (decel: number) => Math.min(time * decel, travel === undefined ? Infinity : Math.sqrt(2 * travel * decel));
54:   const h = (v: number) => (v * v) / (2 * g);
55:   const survive = h(speedFor(p.hurtDecel));
56:   return { clean: style === 'roll' ? survive : h(speedFor(p.stumbleDecel)), survive };
57: }

1: import { DT, GRAVITY } from './units';

3: // The warped wall: a quarter-circle of radius = wall height H, flat at the bottom, vertical at the top.
4: // Speed turns into height; the wall cannot add any. Friction is a share `loss` of the energy spent climbing.
5: export interface WallParams {
6:   height: number; // m, wall top above the run-up ground
7:   radius?: number; // m, curve radius (default: the wall height); the Dojo keeps it constant so height does not change the curve
8:   stopAtGrab?: boolean; // default true: the run ends when the hands reach the top; false follows the climb to its peak
9:   hipHeight: number; // m, hips above the feet while running
10:   reach: number; // m, hands above the hips when stretched up (1.3: about what a body can stretch; was 2.1 in the spec)
11:   loss: number; // friction loss as a share of the climb energy (starting value)
12:   gravity: number;
13: }
14: 
15: export const DEFAULT_WALL: WallParams = { height: 4.3, hipHeight: 1.0, reach: 1.3, loss: 0.1, gravity: GRAVITY };
16: 
17: export interface WallResult {
18:   grabbed: boolean;
19:   heightGained: number; // m, highest the feet got up the curve
20:   needed: number; // m the feet must rise so the hands reach the top: height - hip - reach
21:   margin: number; // heightGained - needed: positive = hands over the edge, negative = short by this much
22:   grabTick: number | null; // tick (from entering the curve) when the hands reached the top
23:   path: { x: number; y: number }[]; // feet position each tick: up the curve, then the slide back if it failed
24:   exitSpeed: number; // m/s on returning to the base (0 if grabbed)
25: }
26: 
27: // Height the feet reach on the curve from entry speed v: v² / (2 g (1 + loss)).
28: export const climbHeight = (v: number, p: WallParams = DEFAULT_WALL): number => (v * v) / (2 * p.gravity * (1 + p.loss));
29: 
30: export function runWall(entrySpeed: number, p: WallParams = DEFAULT_WALL): WallResult {
31:   const R = p.radius ?? p.height;
32:   const arcLen = (Math.PI / 2) * R;
33:   const stopAtGrab = p.stopAtGrab ?? true;
34:   const needed = p.height - p.hipHeight - p.reach;
35:   // Along the arc: s'' = -g sin(s / R) (1 + loss * sign(v)); friction always opposes the motion,
36:   // so going up costs (1 + loss) and sliding down gives back only (1 - loss).
37:   const acc = (pos: number, vel: number) => -p.gravity * Math.sin(Math.min(pos / R, Math.PI / 2)) * (1 + p.loss * Math.sign(vel));
38:   let s = 0;
39:   let v = entrySpeed;
40:   let peak = 0;
41:   let grabTick: number | null = null;
42:   const path: { x: number; y: number }[] = [];
43:   for (let tick = 0; tick < 120 * 30; tick++) {
44:     const th = Math.min(s / R, Math.PI / 2);
45:     const y = R * (1 - Math.cos(th)) + Math.max(0, s - arcLen); // past the arc the wall is vertical
46:     path.push({ x: R * Math.sin(th), y });
47:     peak = Math.max(peak, y);
48:     if (y >= needed && grabTick === null) { grabTick = tick; if (stopAtGrab) break; }
49:     // RK2 (midpoint) on (s, v)
50:     const a1 = acc(s, v);
51:     const sm = s + v * DT / 2;
52:     const vm = v + a1 * DT / 2;
53:     const a2 = acc(sm, vm);
54:     s += vm * DT;
55:     v += a2 * DT;
56:     if (s <= 0) { s = 0; if (v < 0) break; }
57:   }
58:   const grabbed = grabTick !== null;
59:   return { grabbed, heightGained: peak, needed, margin: peak - needed, grabTick, path, exitSpeed: grabbed ? 0 : Math.max(0, -v) };
60: }

14: // A roll is effort like a vault, and the body needs a moment to be ready again: it cannot be spammed.
15: export const GROUND_ROLL_COST = 0.02; // share of a full energy store (a vault is 0.025, a jump 0.06)

20:   start: number; // tick the roll began
21:   end: number; // tick the roll ends (start + GROUND_ROLL_TIME)

41:   const done = tick >= g.end;

12: // Where the runner is about to land and how hard: the fall in the next quarter of a second (null when not falling toward the ground).
13: // The roll is pressed inside a tight window: the last 0.15 s before touchdown AND the last 1.2 m above the ground (both at once, so whichever is shorter).
14: // An earlier press does nothing (there is no hidden buffer: the landing is then a normal one); a press up to 3 ticks after touchdown is forgiven.
15: export const ROLL_WINDOW_S = 0.15;
16: export const ROLL_WINDOW_M = 1.2;
17: export const ROLL_FORGIVE_TICKS = 3;
18: 
19: function comingFall(w: WorldState, course: CourseData): { v: number; tau: number; h: number } | null {
20:   const r = w.runner;
21:   if (r.grounded || w.mode !== 'run' || r.vy >= 0) return null;
22:   const under = (x: number) => Math.max(course.terrain.waterY, groundAt(course.terrain, x)); // over a gap there is no ground: the water is the floor
23:   let floor = under(r.x), tau = 0;
24:   for (let i = 0; i < 8; i++) { // the landing spot and the fall time depend on each other (an edge or a gap below): iterate until they agree
25:     const d = r.y - floor;
26:     tau = d <= 0 ? 0 : (r.vy + Math.sqrt(r.vy * r.vy + 2 * GRAVITY * d)) / GRAVITY; // time to fall d metres from the speed vy (negative is down)
27:     const next = under(r.x + r.vx * tau);
28:     if (next === floor) break;
29:     floor = next;
30:   }
31:   return { v: Math.sqrt(r.vy * r.vy + 2 * GRAVITY * Math.max(0, r.y - floor)), tau, h: Math.max(0, r.y - floor) };
32: }

35: export function rollWindowOpen(w: WorldState, course: CourseData): boolean {
36:   const f = comingFall(w, course);
37:   return !!f && f.tau <= ROLL_WINDOW_S + 1e-9 && f.h <= ROLL_WINDOW_M + 1e-9;
