47: export function makeRails(pts: [number, number][]): Rails {
48:   const p = [...pts].sort((a, b) => a[0] - b[0]);
49:   const n = p.length;
50:   const xs = p.map((q) => q[0]), ys = p.map((q) => q[1]);
51:   const M = new Array<number>(n).fill(0); // second derivatives at the points; zero at both ends (natural)
52:   if (n > 2) {
53:     const h = xs.slice(1).map((x, i) => x - xs[i]);
54:     const a = new Array<number>(n).fill(0), b = new Array<number>(n).fill(1), c = new Array<number>(n).fill(0), d = new Array<number>(n).fill(0);
55:     for (let i = 1; i < n - 1; i++) {
56:       a[i] = h[i - 1]; b[i] = 2 * (h[i - 1] + h[i]); c[i] = h[i];
57:       d[i] = 6 * ((ys[i + 1] - ys[i]) / h[i] - (ys[i] - ys[i - 1]) / h[i - 1]);
58:     }
59:     for (let i = 1; i < n; i++) { const w = a[i] / b[i - 1]; b[i] -= w * c[i - 1]; d[i] -= w * d[i - 1]; } // forward sweep
60:     M[n - 1] = d[n - 1] / b[n - 1];
61:     for (let i = n - 2; i >= 0; i--) M[i] = (d[i] - c[i] * M[i + 1]) / b[i];
62:   }
63:   const seg = (x: number): number => {
64:     let i = 0;
65:     while (i < n - 2 && x > xs[i + 1]) i++;
66:     return i;
67:   };
68:   const clampX = (x: number) => Math.min(Math.max(x, xs[0]), xs[n - 1]);
69:   return {
70:     x0: xs[0], x1: xs[n - 1],
71:     y(x) {
72:       if (n < 2) return ys[0] ?? 0;
73:       x = clampX(x); const i = seg(x), h = xs[i + 1] - xs[i], l = xs[i + 1] - x, r = x - xs[i];
74:       return (M[i] * l ** 3) / (6 * h) + (M[i + 1] * r ** 3) / (6 * h) + (ys[i] / h - (M[i] * h) / 6) * l + (ys[i + 1] / h - (M[i + 1] * h) / 6) * r;
75:     },
76:     slope(x) {
77:       if (n < 2) return 0;
78:       x = clampX(x); const i = seg(x), h = xs[i + 1] - xs[i], l = xs[i + 1] - x, r = x - xs[i];
79:       return (-M[i] * l * l) / (2 * h) + (M[i + 1] * r * r) / (2 * h) - (ys[i] / h - (M[i] * h) / 6) + (ys[i + 1] / h - (M[i + 1] * h) / 6);
80:     },

84:       return (M[i] * (xs[i + 1] - x)) / h + (M[i + 1] * (x - xs[i])) / h;

101: // How hard the rails push up on the cart, per kilogram: the weight across the slope plus the push of a valley, minus the lift of a crest.
102: // At zero or below the cart is no longer held on and leaves the track.
103: export function railPush(rails: Rails, x: number, v: number, g: number = GRAVITY): number {

115: // One RK4 step along the rails. `brake` is held. Returns the new x and speed.
116: export function cartStep(rails: Rails, x: number, v: number, brake: boolean, p: CartParams = DEFAULT_CART): { x: number; v: number } {
117:   const acc = (xx: number, vv: number): number => {
118:     const s = rails.slope(xx), c = 1 / Math.sqrt(1 + s * s), sinPhi = s * c;
119:     const smooth = Math.tanh(vv / 0.05); // resistance opposes the motion, and fades to nothing at rest (no jitter)
120:     return -p.gravity * sinPhi - p.rolling * p.gravity * c * smooth - p.drag * vv * Math.abs(vv) - (brake ? p.brake * smooth : 0);
121:   };
122:   const vel = (xx: number, vv: number): number => vv / Math.sqrt(1 + rails.slope(xx) ** 2);
123:   const k1x = vel(x, v), k1v = acc(x, v);
124:   const k2x = vel(x + 0.5 * DT * k1x, v + 0.5 * DT * k1v), k2v = acc(x + 0.5 * DT * k1x, v + 0.5 * DT * k1v);
125:   const k3x = vel(x + 0.5 * DT * k2x, v + 0.5 * DT * k2v), k3v = acc(x + 0.5 * DT * k2x, v + 0.5 * DT * k2v);
126:   const k4x = vel(x + DT * k3x, v + DT * k3v), k4v = acc(x + DT * k3x, v + DT * k3v);
127:   return { x: x + (DT / 6) * (k1x + 2 * k2x + 2 * k3x + k4x), v: v + (DT / 6) * (k1v + 2 * k2v + 2 * k3v + k4v) };
128: }

7: // The plank bridge: a plank laid across a gap, held at its two ends (and, if asked, by a post in the middle).
8: // The runner is a load that walks along it. The plank bends under the load (it sags) and breaks if the bending is too much.
9: // Euler-Bernoulli beam, simply supported, Earth gravity only; every number is a metre, a second or a kilogram.
10: //   sag under a load F in the middle of a span L:   F L^3 / (48 E I)   (E: how stiff the material is, I: how stiff the cross-section is)
11: //   I of a rectangular plank:                       width x thickness^3 / 12
12: //   bending stress in the outer face:               M (thickness / 2) / I, with M = F L / 4 in the middle
13: // Simplifications (stated, not hidden): the load is the rider's weight applied slowly (no extra push from running or jumping);
14: // a post in the middle turns the plank into two planks that meet on the post (the continuous-beam gain is ignored, so a post helps slightly less than in a real plank);
15: // the plank's own weight is spread evenly; no creep, no knots, no damage from earlier crossings.

52: // Sag (m, downward) at x (m from the left end) with the rider standing at a (m from the left end; a < 0: nobody on the plank), self-weight included.
53: export function sagAt(pk: PlankData, a: number, x: number, mass: number = RIDER_MASS): number {
54:   const EI = materialOf(pk).E * secondMoment(pk);
55:   const seg = segOf(pk, x), l = seg.l, xl = x - seg.s0;
56:   // own weight, spread evenly: q x (l^3 - 2 l x^2 + x^3) / (24 EI)
57:   const q = weightPerMetre(pk);
58:   let d = (q * xl * (l ** 3 - 2 * l * xl * xl + xl ** 3)) / (24 * EI);
59:   // the rider, if on this segment: a point load
60:   if (a >= 0 && a <= spanOf(pk) && segOf(pk, a).s0 === seg.s0) {
61:     const F = mass * GRAVITY, al = a - seg.s0, b = l - al;
62:     d += xl <= al ? (F * b * xl * (l * l - b * b - xl * xl)) / (6 * EI * l) : (F * al * (l - xl) * (l * l - al * al - (l - xl) ** 2)) / (6 * EI * l);
63:   }
64:   return d;
65: }

67: // The largest bending moment (N m) anywhere on the plank with the rider at a.
68: export function maxMoment(pk: PlankData, a: number, mass: number = RIDER_MASS): number {
69:   const q = weightPerMetre(pk), F = mass * GRAVITY, L = spanOf(pk);
70:   let best = 0;
71:   const N = 40;
72:   for (let i = 0; i <= N; i++) {
73:     const x = (L * i) / N;
74:     const seg = segOf(pk, Math.min(x, L - 1e-9)), l = seg.l, xl = x - seg.s0;
75:     let M = (q * xl * (l - xl)) / 2;
76:     if (a >= 0 && segOf(pk, Math.min(a, L - 1e-9)).s0 === seg.s0) {
77:       const al = a - seg.s0, b = l - al;
78:       M += xl <= al ? (F * b * xl) / l : (F * al * (l - xl)) / l;
79:     }
80:     best = Math.max(best, M);
81:   }
82:   return best;
83: }

166: export function stepPlankWorld(w: WorldState, input: RunnerInput, course: CourseData): WorldState {
167:   const st = w.plank!, pk = course.planks![st.i];
168:   const tick = w.runner.tick + 1;
169:   const events: GameEvent[] = [];
170:   const p = DEFAULT_PARAMS;
171:   const tau = p.topSpeed / p.acceleration;
172:   // the legs: the same push as on the ground
173:   let v = st.v;
174:   const run = input.run && !input.left, back = input.left && !input.run;
175:   if (run) v += ((p.topSpeed - v) / tau) * DT;
176:   else if (back) v += ((-p.topSpeed - v) / tau) * DT;
177:   else v -= v * Math.min(1, 3 * DT);
178:   // the dip: climbing out of it costs speed, and a plank that is shaking takes some away
179:   const a0 = w.runner.x - pk.x0;
180:   const slope = (topAt(pk, a0 + 0.05, pk.x0 + a0 + 0.05) - topAt(pk, a0 - 0.05, pk.x0 + a0 - 0.05)) / 0.1;
181:   const sag0 = sagAt(pk, a0, a0);
182:   v += -GRAVITY * Math.sin(Math.atan(slope)) * DT;
183:   v -= v * SAG_LOSS * sag0 * DT;
184:   let x = w.runner.x + v * DT;
185:   const a = x - pk.x0;
186:   // the jump leaves the plank
187:   if (input.jump) {
188:     events.push({ tick, type: 'jump', detail: { vy: course.jumpSpeed, vx: v, x } });

190:   }
191:   // the far end: back on a deck
192:   if (x >= pk.x1 || x <= pk.x0) {
193:     const side = x >= pk.x1 ? pk.x1 : pk.x0;
194:     const deck = standingOn(course.terrain, side + (side === pk.x1 ? 0.02 : -0.02));
195:     events.push({ tick, type: 'plank-off', detail: { x: side } });
196:     return { ...w, mode: 'run', plank: null, events, runner: { ...w.runner, tick, x, y: deck ? deck.top : pk.y, vx: v, vy: 0, grounded: !!deck, landedAtTick: null, landedAtX: null } };
197:   }
198:   // too much bending: it breaks under the feet
199:   const ratio = stressRatio(pk, a);
200:   if (ratio >= 1) {
201:     events.push({ tick, type: 'plank-break', detail: { x, at: a, ratio, sag: sagAt(pk, a, a) } });
202:     return { ...w, mode: 'run', plank: null, broken: [...(w.broken ?? []), st.i], events, runner: { ...w.runner, tick, x, y: topAt(pk, a, x), vx: v, vy: -1.2, grounded: false, landedAtTick: null, landedAtX: null } };
203:   }
204:   const sag = sagAt(pk, a, a);
205:   let sagged = st.sagged;
206:   if (!sagged && sag >= SAG_NOTE) { sagged = true; events.push({ tick, type: 'plank-sag', detail: { x, sag, ratio } }); }

208:   const vy = (y - w.runner.y) / DT;
209:   return { ...w, plank: { i: st.i, v, sagged }, events, runner: { ...w.runner, tick, x, y, vx: v, vy, grounded: true } };

83: export function droneStep(s: DroneState, cmd: DroneCmd, p: DroneParams): DroneState {

85:   const lean = s.lean + (cmd.lean * p.maxLean - s.lean) * (1 - Math.exp(-DT / p.leanLag));

88:   if (s.empty) T = 0;
89:   else if (p.stabiliser) {

92:     const az = clamp(4 * (vz - s.vy), -6, 6);
93:     T = clamp((m * (GRAVITY + az + p.drag * s.vy)) / Math.cos(lean), 0, p.maxThrust);
94:   } else {
95:     throttle = clamp(throttle + cmd.climb * 0.3 * DT, 0, 1);

100:   const ay = (T * Math.cos(lean)) / m - GRAVITY - p.drag * s.vy;

102:   const power = rotorPower(T, p);
103:   const energy = Math.max(0, s.energy - power * DT);
104:   return { t: s.t + DT, x: s.x + vx * DT, y: s.y + vy * DT, vx, vy, lean, hold, throttle, thrust: T, power, energy, empty: s.empty || energy <= 0 };

4: // Recoil: momentum is conserved. A launcher on the runner throws a heavy ball backward; the runner is pushed the other way by exactly

10: //   (m_c + m_b) v = m_c v2 + m_b (v2 + rel)   =>   v2 = v + m_b u (cos a, sin a) / (m_c + m_b),   rel = -u (cos a, sin a)

53: function accel(p: TrebuchetParams, al: number, be: number, dal: number, dbe: number): [number, number] {
54:   const { long: b, short: a, sling: s, counterweight: M, armMass: ma, riderMass: m, gravity: g } = p;
55:   const Iarm = (ma / 12) * (a + b) ** 2 + ma * ((b - a) / 2) ** 2;
56:   const K = -M * a + (ma * (b - a)) / 2 + m * b;
57:   const M11 = Iarm + M * a * a + m * b * b, M12 = m * b * s * Math.cos(be - al), M22 = m * s * s;
58:   const r1 = m * b * s * Math.sin(be - al) * dbe * dbe - g * K * Math.cos(al);
59:   const r2 = -m * b * s * Math.sin(be - al) * dal * dal - m * g * s * Math.cos(be);
60:   const det = M11 * M22 - M12 * M12;

83: // The arm has hit the frame: the tip is fixed and the rider swings round it like a pendulum (beta'' = -(g / sling) cos beta).
84: function stoppedStep(s: Launch, p: TrebuchetParams): Launch {

86:   const k1 = f(s.beta), b2 = s.beta + 0.5 * DT * s.dbeta, d2 = s.dbeta + 0.5 * DT * k1;
87:   const k2 = f(b2), b3 = s.beta + 0.5 * DT * d2, d3 = s.dbeta + 0.5 * DT * k2;
88:   const k3 = f(b3), b4 = s.beta + DT * d3, d4 = s.dbeta + DT * k3;
89:   const k4 = f(b4);

118: // The energy the falling counterweight has given up so far (J, net of the arm's own rise), and the rider's kinetic energy (J):
119: // the throw can never exceed the first (a well-tuned real trebuchet turns about half to two thirds of it into the throw).
120: export function releasedEnergy(p: TrebuchetParams, s: Launch): number {

145: // Total mechanical energy of the machine (J): kinetic plus potential of the arm, the counterweight and the rider. With no friction it is constant.
146: export function mechanicalEnergy(p: TrebuchetParams, s: Launch): number {
147:   const { long: b, short: a, sling: l, counterweight: M, armMass: ma, riderMass: m, gravity: g, axleHeight: H } = p;
148:   const Iarm = (ma / 12) * (a + b) ** 2 + ma * ((b - a) / 2) ** 2;
149:   const M11 = Iarm + M * a * a + m * b * b, M12 = m * b * l * Math.cos(s.beta - s.alpha), M22 = m * l * l;
150:   const T = 0.5 * M11 * s.dalpha ** 2 + M12 * s.dalpha * s.dbeta + 0.5 * M22 * s.dbeta ** 2;
151:   const V = g * (M * (H - a * Math.sin(s.alpha)) + ma * (H + ((b - a) / 2) * Math.sin(s.alpha)) + m * (H + b * Math.sin(s.alpha) + l * Math.sin(s.beta)));
152:   return T + V;
153: }

21:   let windowX: number | null = null; // where the runner was when the roll window opened in this flight
22:   let still = 0;
23:   for (let i = 0; i < maxTicks && w.outcome === 'running'; i++) {
24:     const input = inputFor(plan, w, st, course);
25:     // a press is the first tick the jump input is on after being off (holding it is not a new drum)
