214: export function driveSync<T>(course: CourseData, g: Gen<T>): T {
215:   let r = g.next();
216:   while (!r.done) {
217:     const { jobs, stopOnClear } = r.value;
218:     const out: number[] = [];
219:     for (const j of jobs) { const s = evalJob(course, j); out.push(s); if (stopOnClear && s >= CLEARED) break; }
220:     r = g.next(out);
221:   }
222:   return r.value;
223: }

225: // `evaluate` plays a list of jobs somewhere else and gives back their scores in the same order. A batch that stops at the first clear is asked for in rounds.
226: export async function driveAsync<T>(g: Gen<T>, evaluate: (jobs: Job[]) => Promise<number[]>, round = 48, onBatch?: (b: Batch) => void): Promise<T> {
227:   let r = g.next();
228:   while (!r.done) {
229:     const b = r.value;
230:     onBatch?.(b);
231:     const out: number[] = [];
232:     if (!b.stopOnClear) out.push(...(await evaluate(b.jobs)));
233:     else for (let i = 0; i < b.jobs.length; i += round) {
234:       const got = await evaluate(b.jobs.slice(i, i + round));
235:       const k = got.findIndex((s) => s >= CLEARED);
236:       if (k >= 0) { out.push(...got.slice(0, k + 1)); break; }
237:       out.push(...got);
238:     }
239:     r = g.next(out);
240:   }
241:   return r.value;
242: }

398: // Two ways through a pair of bars (asked for with `{ routes: true }`: it adds about 250 runs, so the stored plans and the tests that need it use it, not every search):
399: // bar to bar (the plan the search found), or a full pump on the first bar and a flight over the second (a skip).
400: // Both are tried and the shorter total time is the line the plan follows; both are reported (the hint and the debug view say what is possible, as a fact).
401: function compareRoutes(course: CourseData, plan: Plan, base: RunResult, windows: Window[]): { plan: Plan; cleared: boolean; routes: Route[] } {
402:   const catches = (r: RunResult) => r.events.filter((e) => e.type === 'hang-catch').length;
403:   const baseBars = catches(base);
404:   type Opt = { plan: Plan; ticks: number; bars: number };
405:   const jx0 = plan.jumpXs[plan.jumpXs.length - 1];
406:   // bar to bar: the fastest way that catches both bars (the plan found may fly over the second one by luck, so the two ways are searched on their own)
407:   let viaBars: Opt | null = baseBars >= 2 ? { plan, ticks: base.world.runner.tick, bars: baseBars } : null;
408:   for (let h1 = 0; h1 <= 300; h1 += 30) for (let h2 = 0; h2 <= 480; h2 += 30) {
409:     const pl: Plan = { ...plan, hangTicks: [h1, h2, ...(plan.hangTicks?.slice(2) ?? [])] };
410:     delete pl.airHands;
411:     const r = playPlan(course, pl, 120 * 40);
412:     if (r.world.outcome === 'cleared' && catches(r) >= 2 && (!viaBars || r.world.runner.tick < viaBars.ticks)) viaBars = { plan: pl, ticks: r.world.runner.tick, bars: catches(r) };
413:   }
414:   // skip: a full pump on the first bar and the hand kept off in the air, so the second bar is flown over
415:   let skip: Opt | null = baseBars === 1 ? { plan, ticks: base.world.runner.tick, bars: 1 } : null;
416:   const skipPlan = (h: number, jx: number): Plan => ({ ...plan, jumpXs: [...plan.jumpXs.slice(0, -1), jx], hangTicks: [h], airHands: [false] });
417:   const tryHang = (jx: number, hs: number[]) => {
418:     for (const h of hs) {
419:       const r = playPlan(course, skipPlan(h, jx), 120 * 40);
420:       if (r.world.outcome === 'cleared' && catches(r) === 1 && (!skip || r.world.runner.tick < skip.ticks)) skip = { plan: skipPlan(h, jx), ticks: r.world.runner.tick, bars: 1 };
421:     }
422:   };
423:   const coarse = Array.from({ length: 91 }, (_, i) => i * 6);
424:   for (const jx of [jx0, ...[-0.3, -0.2, -0.1, 0.1, 0.2, 0.3].map((d) => Math.round((jx0 + d) * 100) / 100)]) { tryHang(jx, coarse); if (skip && skip.plan.airHands) break; }
425:   if (skip && skip.plan.airHands) { const b = skip as Opt; tryHang(b.plan.jumpXs[b.plan.jumpXs.length - 1], Array.from({ length: 13 }, (_, i) => Math.max(0, b.plan.hangTicks![0] - 6 + i))); }
426:   const opts = [viaBars && { name: 'bar to bar', ...viaBars }, skip && { name: 'skip the second bar', ...(skip as Opt) }].filter((o): o is Opt & { name: string } => !!o);
427:   if (!opts.length) return { plan, cleared: true, routes: [{ name: baseBars >= 2 ? 'bar to bar' : 'one bar', ticks: base.world.runner.tick, bars: baseBars, chosen: true }] };
428:   const fastest = opts.reduce((a, b) => (b.ticks < a.ticks ? b : a));
429:   const routes: Route[] = opts.map((o) => ({ name: o.name, ticks: o.ticks, bars: o.bars, chosen: o === fastest }));
430:   if (fastest.name.startsWith('skip')) windows.push({ obstacle: 'skip the second bar', best: fastest.plan.hangTicks![0], lo: fastest.plan.hangTicks![0], hi: fastest.plan.hangTicks![0], widthMs: DT * 1000 });
431:   return { plan: fastest.plan, cleared: true, routes };
432: }

457: // The launcher: when to fire (at a place on the ground, or some ticks after the take-off) and when to jump, found together because they depend on each other.
458: function searchRecoil(course: CourseData, plan: Plan, windows: Window[], score: (r: RunResult) => number): number | null {
459:   const rp = recoilParams(course);
460:   const e = edges(course).find((q) => q.x > rp.x1);

463:   type T = { sc: number; jx: number; fire: { x?: number; air?: number } };
464:   const mk = (jx: number, fire: { x?: number; air?: number }): Plan => ({ ...plan, jumpXs: [...plan.jumpXs, jx], ...(fire.x !== undefined ? { fireXs: [fire.x] } : {}), ...(fire.air !== undefined ? { fireAfterJump: fire.air } : {}) });
465:   const trial = (jx: number, fire: { x?: number; air?: number }): T => ({ sc: score(playPlan(course, mk(jx, fire), 120 * 30)), jx, fire });
466:   const found: T[] = [];
467:   const jxs = candidates(e.x - 3, e.x, 0.1);
468:   for (const jx of jxs) for (let a = 0; a <= 60; a += 6) found.push(trial(jx, { air: a })); // a shot in the air, a moment after the take-off
469:   for (const jx of jxs) for (let d = 0; d <= 2; d += 0.25) found.push(trial(jx, { x: Math.max(rp.x0, jx - d) })); // or a shot on the ground just before it
470:   const tops = [...found].sort((a, b) => b.sc - a.sc).slice(0, 8);
471:   for (const t of tops) for (let jx = t.jx - 0.1; jx <= t.jx + 0.1; jx += 0.02) found.push(trial(Math.round(jx * 1000) / 1000, t.fire));
472:   const best = Math.max(...found.map((f) => f.sc));
473:   const good = found.filter((f) => f.sc >= best - 1e-9);
474:   const air = good.filter((g) => g.fire.air !== undefined), ground = good.filter((g) => g.fire.x !== undefined);
475:   const family = air.length >= ground.length ? air : ground; // the family with the wider window of good presses
476:   const mid = [...family].sort((a, b) => a.jx - b.jx)[Math.floor(family.length / 2)];
477:   const jxsGood = family.map((g) => g.jx);
478:   plan.jumpXs.push(mid.jx);
479:   if (mid.fire.x !== undefined) plan.fireXs = [mid.fire.x]; else plan.fireAfterJump = mid.fire.air;
480:   windows.push({ obstacle: `jump at ${e.name}`, best: mid.jx, lo: Math.min(...jxsGood), hi: Math.max(...jxsGood), widthMs: ((Math.max(...jxsGood) - Math.min(...jxsGood) + 0.02) / 7.4) * 1000 });

509:   const trial = (st: number, jx: number) => { const pl: Plan = { ...plan, bandStretch: st, jumpXs: [...plan.jumpXs, jx] }; return { sc: score(playPlan(course, pl, 120 * 40)), st, jx }; };
510:   const found: { sc: number; st: number; jx: number }[] = [];
511:   for (let st = Math.min(1, top); st <= top + 1e-9; st += top > 8 ? 1 : 0.5) for (const jx of candidates(e.x - 2.5, e.x, 0.25)) found.push(trial(st, jx));
512:   const tops = [...found].sort((a, b) => b.sc - a.sc).slice(0, 6);

557: // A player who is good but not perfect: every press is a little off (sigma in ms of game time), and a fall
558: // means starting again. Returns the game seconds until the first clear, or null if it never clears in `limitS`.
559: export function playImperfect(course: CourseData, plan: Plan, seed: number, sigmaMs: number, limitS = 180): number | null {
560:   let state = seed >>> 0 || 1;
561:   const rnd = () => { state = (Math.imul(state, 1664525) + 1013904223) >>> 0; return state / 2 ** 32; };
562:   const gauss = () => Math.sqrt(-2 * Math.log(Math.max(rnd(), 1e-12))) * Math.cos(2 * Math.PI * rnd());

569:       bouncePress: Math.max(0, Math.round(plan.bouncePress + (gauss() * sigmaMs) / (DT * 1000))),

5: // How forgiving is the release of a trebuchet course? The window of release times (ticks after the pin is pulled) that still win,
6: // measured from the simulator: for every release time near the best one, the player may hold on to the bar for as long as they like
7: // (the cue tells them when to let go), so a release time counts when any hang time clears. The window is the unbroken run around the best release.
8: export interface ReleaseWindow { best: number; lo: number; hi: number; ticks: number; widthMs: number }
9: 
10: export function releaseWindow(course: CourseData, found?: { plan: Plan; cleared: boolean }): ReleaseWindow | null {
11:   const r = found ?? searchIdealRun({ ...course, plan: undefined });
12:   if (!r.cleared || r.plan.launchRelease === undefined) return null;
13:   const plan = r.plan, best = plan.launchRelease!;
14:   const bars = (course.holds ?? []).length > 0;
15:   const hangs = bars ? Array.from({ length: 81 }, (_, i) => i * 6) : [0];
16:   const wins = (rel: number): boolean => {
17:     const base: Plan = { ...plan, launchRelease: rel };
18:     if (playPlan(course, base, 120 * 30).world.outcome === 'cleared') return true; // the stored hang time first: nearly always enough
19:     if (!bars) return false;
20:     return hangs.some((h) => playPlan(course, { ...base, hangTicks: plan.hangTicks!.map((t, i) => (i === 0 ? h : t)) }, 120 * 30).world.outcome === 'cleared');
21:   };
22:   let lo = best, hi = best;
23:   while (lo > 20 && wins(lo - 1)) lo--;
24:   while (hi < 600 && wins(hi + 1)) hi++;
25:   return { best, lo, hi, ticks: hi - lo + 1, widthMs: (hi - lo + 1) * DT * 1000 };
26: }
